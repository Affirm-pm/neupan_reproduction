find_package(generate_parameter_library REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(OsqpEigen REQUIRED)
get_filename_component(NEUPAN_BUNDLED_SOURCE_DIR
  "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../neupan_cpp_ws/src/neupan_cpp/libneupan" ABSOLUTE)
set(NEUPAN_SOURCE_DIR "${NEUPAN_BUNDLED_SOURCE_DIR}" CACHE PATH "Path to libneupan source")
if(NOT EXISTS "${NEUPAN_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set -DNEUPAN_SOURCE_DIR=/path/to/neupan_cpp/libneupan")
endif()
set(NEUPAN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NEUPAN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(${NEUPAN_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/libneupan)
# Ship the two small solver DSOs alongside the plugin. Sourcing this package is
# sufficient at runtime; no separate dependency workspace needs to be sourced.
get_target_property(NEUPAN_OSQP_EIGEN_LIB OsqpEigen::OsqpEigen IMPORTED_LOCATION_RELEASE)
get_target_property(NEUPAN_OSQP_LIB osqp::osqp IMPORTED_LOCATION_RELEASE)
install(FILES ${NEUPAN_OSQP_EIGEN_LIB} ${NEUPAN_OSQP_LIB} DESTINATION lib)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/neupan_library_path.dsv "prepend-non-duplicate;LD_LIBRARY_PATH;lib\n")
ament_environment_hooks(${CMAKE_CURRENT_BINARY_DIR}/neupan_library_path.dsv)
foreach(name neupan_planner neupan_cloud_map neupan_world_view)
  generate_parameter_library(${name}_params ${name}.yaml)
endforeach()
add_library(finenav_neupan_plugin SHARED src/neupan_plugin.cpp)
target_include_directories(finenav_neupan_plugin PUBLIC include)
ament_target_dependencies(finenav_neupan_plugin finenav_core sensor_msgs ament_index_cpp)
target_link_libraries(finenav_neupan_plugin neupan neupan_planner_params neupan_cloud_map_params neupan_world_view_params)
install(TARGETS finenav_neupan_plugin LIBRARY DESTINATION lib)
install(DIRECTORY models behavior_trees tools DESTINATION share/${PROJECT_NAME}/create_custom_neupan_planner)
install(DIRECTORY gazebo DESTINATION share/${PROJECT_NAME}/create_custom_neupan_planner
  PATTERN "__pycache__" EXCLUDE)
install(PROGRAMS gazebo/neupan_gazebo_adapter.py DESTINATION lib/${PROJECT_NAME})
install(FILES planner.yaml neupan_sim.yaml README.md VALIDATION.md LICENSE DESTINATION share/${PROJECT_NAME}/create_custom_neupan_planner)
install(FILES test/adapter_result.xml test/smoke_result.json test/smoke_engine.log
  test/current_robot_adapter_result.xml
  DESTINATION share/${PROJECT_NAME}/create_custom_neupan_planner/test)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_neupan_adapter test/test_adapter.cpp)
  target_link_libraries(test_neupan_adapter finenav_neupan_plugin)
  target_compile_definitions(test_neupan_adapter PRIVATE NEUPAN_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
endif()
